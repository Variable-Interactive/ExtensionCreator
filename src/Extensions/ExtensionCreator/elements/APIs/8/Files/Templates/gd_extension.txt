## Automatically extracts dependencies, once the extension loads up (you don't have to touch it)
class DependencyManager:
	var to_path = "user://extensions/<GDExtension>"
	var libraries_zipped_path: String

	func _init(zip_path: String):
		libraries_zipped_path = zip_path

	func extract_deps() -> bool:
		var zip_reader := ZIPReader.new()
		var err := zip_reader.open(libraries_zipped_path)
		if err == OK:
			for path in zip_reader.get_files():
				var content = zip_reader.read_file(path)
				if content:
					DirAccess.make_dir_recursive_absolute(to_path.path_join(path).get_base_dir())
					var file = FileAccess.open(to_path.path_join(path), FileAccess.WRITE)
					file.store_buffer(content)
					file.close()
			return true
		else:
			return false
